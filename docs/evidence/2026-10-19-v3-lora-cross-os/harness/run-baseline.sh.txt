#!/bin/bash
# Baseline: managed runner + two Linux CLI workers on the 5070 Ti.
set -u
cd /home/pc5070ti/projects/heteroes/runtime
D=/home/pc5070ti/projects/heteroes/.scratch/v3-cross-os; R=$D/${1:-linux-baseline}; MP=/home/pc5070ti/.cache/huggingface/hub/models--Qwen--Qwen2.5-0.5B-Instruct/snapshots/7ae557604adf67be50417f59c2c2f167def9a775; PY=$HOME/miniforge3/envs/heteroes-match/bin/python
export PYTHONPATH=src HETEROES_GPU_ACCEPTANCE=1 OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1 MKL_NUM_THREADS=1
$PY $D/harness.py runner --out $R --token-file $D/token > $R/runner-output.txt 2>&1 &
RUNNER=$!
for i in $(seq 600); do grep -q PREPARING $R/runner.json 2>/dev/null && break; sleep 1; done
grep -q PREPARING $R/runner.json || { echo runner-not-preparing; kill $RUNNER; exit 1; }
PIDS=""
for w in linux-a linux-b; do
  HETEROES_TOKEN=$(cat $D/token) $PY scripts/run_worker.py --model-path $MP --model-manifest $R/model-manifest.json --dataset $R/dataset.json --run-config $R/config.json --coordinator-url http://127.0.0.1:18771 --worker-id $w --device cuda --log $R/$w.jsonl --cache-dir $R/cache-$w --replay always > $R/$w-output.txt 2>&1 &
  PIDS="$PIDS $!"
done
for i in $(seq 600); do n=$(grep -l executor_ready $R/linux-*.jsonl 2>/dev/null | wc -l); [ "$n" = 2 ] && break; sleep 1; done
touch $R/start.flag
for p in $PIDS; do wait $p; echo "worker $p exit $?"; done
touch $R/workers-stopped.flag
wait $RUNNER; echo "runner exit $?"
