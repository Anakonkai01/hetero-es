"""Summarize the Linux vs native Windows seed-7 LoRA adapter difference and the CPU uniform_ probe outputs."""
import hashlib, json, math, sys
from pathlib import Path
import numpy as np

d = Path(sys.argv[1])
sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()  # noqa: E731
lin, win = np.fromfile(d / "linux.bin", np.float16), np.fromfile(d / "windows.bin", np.float16)
diff = np.nonzero(lin.view(np.uint16) != win.view(np.uint16))[0]
ulps = np.abs(lin.view(np.uint16)[diff].astype(int) - win.view(np.uint16)[diff].astype(int))
fp32 = np.fromfile(d / "linux-fp32.bin", np.float32)
lu, wu = np.fromfile(d / "linux-uniform.bin", np.float32), np.fromfile(d / "windows-uniform.bin", np.float32)
b = 1 / math.sqrt(896)
f, t = np.float32(-b), np.float32(b)
r32, r64 = np.float32(t - f), float(t) - float(f)
in_double = lambda u: np.float32(u * 2.0**-24 * r64 + float(f))  # noqa: E731
in_float = lambda u: np.float32(np.float32(np.float32(u * 2.0**-24) * r32) + f)  # noqa: E731
explained = 0
for i in range(len(lu)):
    est = int(round((float(lu[i]) - float(f)) / r64 * 2**24))
    window = range(est - 8, est + 9)
    if {u for u in window if in_double(u) == lu[i]} & {u for u in window if in_float(u) == wu[i]}:
        explained += 1
out = {
    "adapter": {
        "linuxSha256": sha(d / "linux.bin"),
        "windowsSha256": sha(d / "windows.bin"),
        "elements": int(len(lin)),
        "nonzeroLinux": int(np.count_nonzero(lin)),
        "nonzeroWindows": int(np.count_nonzero(win)),
        "differingElements": int(len(diff)),
        "fp16UlpDistances": sorted({int(x) for x in ulps}),
        "linuxFp32CastEqualsLinuxFile": bool(np.array_equal(fp32.astype(np.float16).view(np.uint16), lin.view(np.uint16))),
    },
    "uniformProbe": {
        "shape": [4, 896],
        "bound": "1/sqrt(896)",
        "differingFloat32Values": int(np.count_nonzero(lu.view(np.uint32) != wu.view(np.uint32))),
        "maxAbsDifference": float(np.abs(lu - wu).max()),
        "maxAbsDifferenceInUlpOfBound": float(np.abs(lu - wu).max() / np.spacing(np.float32(b))),
        "valuesExplainedBySameDrawLinuxDoubleWindowsFloat": explained,
        "values": int(len(lu)),
    },
}
json.dump(out, sys.stdout, indent=2)
